cmake_minimum_required(VERSION 3.24)
project(PortraitCameraNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SOFTCAM "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/softcam/src")
set(BASE_SOURCES amfilter amvideo combase dllentry dllsetup mtype source winutil wxdebug wxlist wxutil)
foreach(NAME IN LISTS BASE_SOURCES)
  list(APPEND BASE_FILES "${SOFTCAM}/baseclasses/${NAME}.cpp")
endforeach()
file(COPY "${SOFTCAM}/softcamcore" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
file(GLOB CORE_FILES "${CMAKE_CURRENT_BINARY_DIR}/softcamcore/*.cpp")
# CMake already defines _WINDLL for a shared library. Guard the upstream
# definition in a generated copy without modifying the pinned submodule.
file(READ "${SOFTCAM}/baseclasses/wxdebug.cpp" WXDEBUG)
string(REPLACE "#define _WINDLL" "#ifndef _WINDLL\n#define _WINDLL\n#endif" WXDEBUG "${WXDEBUG}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/portrait_wxdebug.cpp" "${WXDEBUG}")
list(REMOVE_ITEM BASE_FILES "${SOFTCAM}/baseclasses/wxdebug.cpp")
list(APPEND BASE_FILES "${CMAKE_CURRENT_BINARY_DIR}/portrait_wxdebug.cpp")
# Use generated branded copies; leave the upstream checkout pristine.
file(READ "${SOFTCAM}/softcam/softcam.cpp" FILTER)
string(REPLACE "0xaef3b972, 0x5fa5, 0x4647, 0x95, 0x71, 0x35, 0x8e, 0xb4, 0x72, 0xbc, 0x9e" "0x89084f72, 0x48e3, 0x4cc4, 0x93, 0xb5, 0x4f, 0x0e, 0x81, 0x55, 0xea, 0x71" FILTER "${FILTER}")
string(REPLACE "DirectShow Softcam" "Portrait Camera (Legacy)" FILTER "${FILTER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/portrait_filter.cpp" "${FILTER}")
file(READ "${SOFTCAM}/softcamcore/FrameBuffer.cpp" FRAMEBUFFER)
string(REPLACE "DirectShow Softcam/" "PortraitCamera-89084f72/v2/" FRAMEBUFFER "${FRAMEBUFFER}")
# A filter may be instantiated only to query formats. Move the receiver heartbeat
# from object lifetime to actual frame delivery, and expose a monotonic counter.
string(FIND "${FRAMEBUFFER}" "        fb.m_receiver_watchdog = Watchdog::createHeartbeat(" RECEIVER_START)
string(FIND "${FRAMEBUFFER}" "        frame->m_watchdog_receiver_heartbeat += 1;\n    }" RECEIVER_END)
if(RECEIVER_START LESS 0 OR RECEIVER_END LESS 0)
  message(FATAL_ERROR "Softcam receiver adaptation no longer matches upstream")
endif()
string(LENGTH "        frame->m_watchdog_receiver_heartbeat += 1;\n" RECEIVER_SUFFIX)
math(EXPR RECEIVER_END "${RECEIVER_END}+${RECEIVER_SUFFIX}")
string(SUBSTRING "${FRAMEBUFFER}" 0 ${RECEIVER_START} FRAMEBUFFER_BEFORE)
string(SUBSTRING "${FRAMEBUFFER}" ${RECEIVER_END} -1 FRAMEBUFFER_AFTER)
set(FRAMEBUFFER "${FRAMEBUFFER_BEFORE}${FRAMEBUFFER_AFTER}")
string(REPLACE "    uint64_t    m_frame_counter;" "    uint64_t    m_frame_counter;\n    uint64_t    m_requested_frames;" FRAMEBUFFER "${FRAMEBUFFER}")
string(REPLACE "        frame->m_frame_counter = 0;" "        frame->m_frame_counter = 0;\n        frame->m_requested_frames = 0;" FRAMEBUFFER "${FRAMEBUFFER}")
string(REPLACE "    auto frame = header();\n    {\n        int w" "    auto frame = header();\n    frame->m_connected_min_version = ProtocolVersion;\n    ++frame->m_watchdog_receiver_heartbeat;\n    ++frame->m_requested_frames;\n    {\n        int w" FRAMEBUFFER "${FRAMEBUFFER}")
string(APPEND FRAMEBUFFER "\nuint64_t softcam::FrameBuffer::requestedFrames() const { std::lock_guard<NamedMutex> guard(m_mutex); return m_shmem ? header()->m_requested_frames : 0; }\n")
file(READ "${SOFTCAM}/softcamcore/FrameBuffer.h" FRAMEBUFFER_HEADER)
string(REPLACE "    uint64_t        frameCounter() const;" "    uint64_t        frameCounter() const;\n    uint64_t        requestedFrames() const;" FRAMEBUFFER_HEADER "${FRAMEBUFFER_HEADER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/softcamcore/FrameBuffer.h" "${FRAMEBUFFER_HEADER}")
file(READ "${SOFTCAM}/softcamcore/SenderAPI.cpp" SENDER)
string(APPEND SENDER "\nextern \"C\" __declspec(dllexport) unsigned long long pcRequestedFrames(void* camera) { auto target = static_cast<Camera*>(camera); return target && s_camera.load() == target ? target->m_frame_buffer.requestedFrames() : 0; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/softcamcore/SenderAPI.cpp" "${SENDER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/portrait_framebuffer.cpp" "${FRAMEBUFFER}")
list(REMOVE_ITEM CORE_FILES "${CMAKE_CURRENT_BINARY_DIR}/softcamcore/FrameBuffer.cpp")
add_library(PortraitCamera.Native SHARED ${BASE_FILES} ${CORE_FILES}
  "${CMAKE_CURRENT_BINARY_DIR}/portrait_filter.cpp" "${CMAKE_CURRENT_BINARY_DIR}/portrait_framebuffer.cpp"
  devices.cpp exports.def)
target_include_directories(PortraitCamera.Native PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/softcamcore" "${SOFTCAM}" "${SOFTCAM}/baseclasses" "${SOFTCAM}/softcam")
target_compile_definitions(PortraitCamera.Native PRIVATE UNICODE _UNICODE WIN32 _WINDOWS _CRT_SECURE_NO_WARNINGS)
target_compile_options(PortraitCamera.Native PRIVATE /W3 /wd4312 /wd4311)
target_link_libraries(PortraitCamera.Native PRIVATE winmm strmiids ole32 oleaut32 uuid advapi32 user32 gdi32)
add_executable(PortraitCamera.Registration registration.cpp)
target_link_libraries(PortraitCamera.Registration PRIVATE strmiids ole32 user32)

  target_compile_features(PortraitCamera.MediaSource PRIVATE cxx_std_20)
  target_include_directories(PortraitCamera.MediaSource PRIVATE mf "../third_party/wil/include"
    "C:/Program Files (x86)/Windows Kits/10/Include/${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}/cppwinrt")

    ole32 oleaut32 propsys shlwapi advapi32 user32 runtimeobject d3d11 dxgi d2d1 dwrite windowscodecs)

  target_compile_features(PortraitCamera.Windows PRIVATE cxx_std_17)
  target_link_libraries(PortraitCamera.Windows PRIVATE mfplat mfuuid mfsensorgroup ole32 advapi32)
